cmake_minimum_required(VERSION 3.24)
project(yolopdf VERSION 0.1.0 LANGUAGES C CXX)
if(MSVC)
    add_compile_options(/utf-8)
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_AUTOMOC ON)
option(YOLOPDF_BUILD_APP "Build the Qt Quick application" ON)
option(YOLOPDF_ANDROID_TEST_HOOKS "Enable Android-only test argument hooks and provider (never ship)" OFF)
set(YOLOPDF_MUPDF_JOBS 4 CACHE STRING "Parallel jobs for the bundled MuPDF build")

find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Concurrent)
include(cmake/BundleMuPDF.cmake)
add_library(yolopdf_engine STATIC src/engine/mupdf/Document.cpp src/engine/mupdf/Content.cpp src/engine/mupdf/Operations.cpp src/engine/mupdf/Render.cpp src/engine/mupdf/Journal.cpp src/engine/mupdf/Lines.cpp src/engine/mupdf/Wrap.cpp)
target_include_directories(yolopdf_engine PUBLIC src)
target_include_directories(yolopdf_engine PRIVATE third_party/mupdf/thirdparty/freetype/include)
target_link_libraries(yolopdf_engine PRIVATE yolopdf_mupdf Qt6::Gui)
if(MSVC)
    target_compile_options(yolopdf_engine PRIVATE /W4 /utf-8)
else()
    target_compile_options(yolopdf_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(ANDROID)
    add_executable(yolopdf-cli tools/yolopdf-cli/native_main.cpp)
    target_link_libraries(yolopdf-cli PRIVATE yolopdf_engine)
else()
    add_executable(yolopdf-cli tools/yolopdf-cli/main.cpp src/platform/FontLocator.cpp)
    target_link_libraries(yolopdf-cli PRIVATE yolopdf_engine Qt6::Core Qt6::Gui)
    if(WIN32)
        target_link_libraries(yolopdf-cli PRIVATE dwrite)
    endif()
endif()

add_library(yolopdf_core STATIC src/core/DocumentWorker.cpp src/core/AssetLibrary.cpp src/core/LibraryArchive.cpp src/platform/FileAccess.cpp src/platform/Print.cpp)
target_include_directories(yolopdf_core PUBLIC src)
target_link_libraries(yolopdf_core PUBLIC yolopdf_engine Qt6::Core Qt6::Gui Qt6::Concurrent)

if(YOLOPDF_BUILD_APP)
    find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 Concurrent)
    qt_standard_project_setup(REQUIRES 6.8)
    qt_add_executable(yolopdf src/app/main.cpp src/app/DocumentController.cpp
        src/app/DocumentController.h src/platform/FontLocator.cpp)
    qt_add_resources(yolopdf stage1_sample PREFIX "/fixtures" BASE src/app/fixtures
        FILES src/app/fixtures/stage1-sample.pdf)
    qt_add_qml_module(yolopdf URI YoloPdf VERSION 1.0
        QML_FILES src/ui/qml/Main.qml)
    target_link_libraries(yolopdf PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Concurrent yolopdf_core)
    if(WIN32)
        target_link_libraries(yolopdf PRIVATE dwrite)
    endif()
    if(NOT ANDROID)
        find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets PrintSupport)
        target_compile_definitions(yolopdf PRIVATE YOLOPDF_DESKTOP_PRINT)
        target_compile_definitions(yolopdf_core PRIVATE YOLOPDF_DESKTOP_PRINT)
        target_link_libraries(yolopdf PRIVATE Qt6::Widgets)
        target_link_libraries(yolopdf_core PRIVATE Qt6::Widgets Qt6::PrintSupport)
    endif()
    if(ANDROID)
        set(android_package "${CMAKE_SOURCE_DIR}/android")
        if(YOLOPDF_ANDROID_TEST_HOOKS)
            set(android_package "${CMAKE_BINARY_DIR}/android-test-package")
            file(COPY "${CMAKE_SOURCE_DIR}/android/" DESTINATION "${android_package}")
            file(COPY "${CMAKE_SOURCE_DIR}/tests/android/Stage1DocumentsProvider.java"
                DESTINATION "${android_package}/src/org/yolopdf")
            file(READ "${android_package}/AndroidManifest.xml" manifest)
            string(REPLACE "</application>"
                "<provider android:name=\"org.yolopdf.Stage1DocumentsProvider\" android:authorities=\"org.yolopdf.app.stage1\" android:exported=\"false\" android:grantUriPermissions=\"true\" /></application>"
                manifest "${manifest}")
            file(WRITE "${android_package}/AndroidManifest.xml" "${manifest}")
            target_compile_definitions(yolopdf PRIVATE YOLOPDF_ANDROID_TEST_HOOKS)
        endif()
        set_property(TARGET yolopdf PROPERTY QT_ANDROID_PACKAGE_SOURCE_DIR "${android_package}")
        set_property(TARGET yolopdf PROPERTY QT_ANDROID_MIN_SDK_VERSION 28)
        set_property(TARGET yolopdf PROPERTY QT_ANDROID_TARGET_SDK_VERSION 35)
    endif()
endif()

include(CTest)
if(BUILD_TESTING AND NOT ANDROID)
    find_package(Qt6 6.8 REQUIRED COMPONENTS Test)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    set(corpus_dir "${CMAKE_BINARY_DIR}/tests/corpus")
    add_custom_command(OUTPUT "${corpus_dir}/.generated"
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/corpus/generate.py" "${corpus_dir}"
        COMMAND "${CMAKE_COMMAND}" -E touch "${corpus_dir}/.generated"
        DEPENDS tests/corpus/generate.py VERBATIM)
    add_custom_target(yolopdf_corpus DEPENDS "${corpus_dir}/.generated")
    add_executable(engine-tests tests/engine/DocumentTest.cpp)
    add_dependencies(engine-tests yolopdf_corpus)
    target_link_libraries(engine-tests PRIVATE yolopdf_engine Qt6::Test)
    target_compile_definitions(engine-tests PRIVATE YOLOPDF_CORPUS_DIR="${corpus_dir}")
    add_test(NAME engine-contract COMMAND engine-tests)
    add_executable(stage1-tests tests/engine/Stage1Test.cpp)
    add_dependencies(stage1-tests yolopdf_corpus)
    target_link_libraries(stage1-tests PRIVATE yolopdf_engine Qt6::Test)
    target_compile_definitions(stage1-tests PRIVATE YOLOPDF_CORPUS_DIR="${corpus_dir}")
    add_test(NAME stage1-contract COMMAND stage1-tests)
    add_executable(session-tests tests/core/SessionTest.cpp)
    add_dependencies(session-tests yolopdf_corpus)
    target_link_libraries(session-tests PRIVATE yolopdf_core Qt6::Test)
    target_compile_definitions(session-tests PRIVATE YOLOPDF_CORPUS_DIR="${corpus_dir}")
    add_test(NAME session-contract COMMAND session-tests)
    set_tests_properties(stage1-contract PROPERTIES TIMEOUT 120)
    add_test(NAME stage1-independent COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_SOURCE_DIR}/tests/engine/test_stage1.py" "$<TARGET_FILE:yolopdf-cli>" "${corpus_dir}")
    set_tests_properties(stage1-independent PROPERTIES TIMEOUT 180)
    if(UNIX)
        add_executable(yolopdf-native-cli tools/yolopdf-cli/native_main.cpp)
        target_link_libraries(yolopdf-native-cli PRIVATE yolopdf_engine)
        add_test(NAME native-cli-replay COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_SOURCE_DIR}/tests/engine/test_native_cli.py" "$<TARGET_FILE:yolopdf-native-cli>" "${corpus_dir}")
    endif()
    add_test(NAME cli-contract COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_SOURCE_DIR}/tests/engine/test_cli.py" "$<TARGET_FILE:yolopdf-cli>" "${corpus_dir}")
    add_test(NAME engine-boundary COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_SOURCE_DIR}/tests/engine/check_boundary.py" "${CMAKE_SOURCE_DIR}")
    if(YOLOPDF_BUILD_APP)
        add_test(NAME app-startup COMMAND "${CMAKE_COMMAND}" -E env
            QT_QPA_PLATFORM=offscreen QT_QUICK_BACKEND=software "$<TARGET_FILE:yolopdf>" --smoke-test)
        set_tests_properties(app-startup PROPERTIES TIMEOUT 20)
        add_test(NAME app-engine-spike COMMAND "${CMAKE_COMMAND}" -E env
            QT_QPA_PLATFORM=offscreen QT_QUICK_BACKEND=software "$<TARGET_FILE:yolopdf>" --smoke-test --fixture-spike)
        set_tests_properties(app-engine-spike PROPERTIES TIMEOUT 30)
    endif()
endif()
